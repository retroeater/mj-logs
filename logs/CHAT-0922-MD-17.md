`.github/workflows/sync-logs.yml`

````yaml
name: 作業ログを mj-logs へ写す

# docs/logs/ の作業ログを public リポジトリ retroeater/mj-logs の logs/ へ写す(#440)。
# mj は private のため、チャット側が報告を貼ってもらわずにログを読めるようにする。
#
# - 写すのは、その push で追加・更新された docs/logs/CHAT-*.md だけ。過去のログは一括で写さない
#   (公開してよいかを点検していないため)。work/** では cloudflare に入っていないコミットのログに限る
#   (cloudflare を取り込んだ merge や早送りで、他のログを写さないため)
# - cloudflare の push では、cloudflare の履歴で削除され現存しないログを mj-logs からも消す。
#   cleanup-logs.yml の削除は GITHUB_TOKEN の push でここを起動しないため、次の cloudflare への push でまとめて消える
# - 書き込みは Actions のシークレット MJ_LOGS_TOKEN(mj-logs にだけ書ける)。値は出力しない

on:
  push:
    branches:
      - cloudflare
      - 'work/**'
    paths:
      - 'docs/logs/**'

permissions:
  contents: read

# mj-logs の main への push を1本ずつにする。待ちは1本までで、さらに来ると古い待ちは取り消される
concurrency:
  group: sync-logs
  cancel-in-progress: false

jobs:
  sync:
    runs-on: ubuntu-latest
    steps:
      - name: リポジトリをチェックアウト
        uses: actions/checkout@v4
        with:
          # push の範囲の差分と、cloudflare の削除履歴を見るため全履歴が要る
          fetch-depth: 0

      - name: mj-logs をチェックアウト
        uses: actions/checkout@v4
        with:
          repository: retroeater/mj-logs
          token: ${{ secrets.MJ_LOGS_TOKEN }}
          path: mj-logs

      - name: ログを写す・消す
        env:
          BEFORE: ${{ github.event.before }}
          AFTER: ${{ github.sha }}
          REF_NAME: ${{ github.ref_name }}
        run: |
          set -euo pipefail
          LOG_RE='^docs/logs/CHAT-[^/]+\.md$'
          # 新規ブランチ(before が 0...)や force push で before が無いときは cloudflare との分岐点から
          if git cat-file -e "$BEFORE^{commit}" 2>/dev/null; then
            BASE="$BEFORE"
          else
            BASE=$(git merge-base origin/cloudflare "$AFTER")
          fi
          if [ "$REF_NAME" = "cloudflare" ]; then
            CHANGED=$(git diff --no-renames --name-only --diff-filter=AM "$BASE" "$AFTER" -- docs/logs)
          else
            CHANGED=$(git log --no-renames --format= --name-only --diff-filter=AM "$BASE..$AFTER" ^origin/cloudflare -- docs/logs)
          fi
          mkdir -p mj-logs/logs
          echo "$CHANGED" | { grep -E "$LOG_RE" || true; } | sort -u | while read -r f; do
            git cat-file -e "$AFTER:$f" 2>/dev/null || continue
            git show "$AFTER:$f" > "mj-logs/logs/$(basename "$f")"
            echo "copy: $f"
          done
          if [ "$REF_NAME" = "cloudflare" ]; then
            git log --no-renames --format= --name-only --diff-filter=D "$AFTER" -- docs/logs \
              | { grep -E "$LOG_RE" || true; } | sort -u | while read -r f; do
              git cat-file -e "$AFTER:$f" 2>/dev/null && continue
              [ -f "mj-logs/logs/$(basename "$f")" ] || continue
              rm "mj-logs/logs/$(basename "$f")"
              echo "delete: $f"
            done
          fi

      - name: mj-logs へ push
        env:
          AFTER: ${{ github.sha }}
          REF_NAME: ${{ github.ref_name }}
        working-directory: mj-logs
        run: |
          set -euo pipefail
          git add -A logs
          if git diff --cached --quiet; then
            echo "変更なし"
            exit 0
          fi
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          {
            echo "sync: ${REF_NAME} ${AFTER:0:8}"
            echo
            echo "retroeater/mj@${AFTER} (${REF_NAME})"
            echo
            git diff --cached --name-status
          } > ../commit_message.txt
          git commit -q -F ../commit_message.txt
          git log -1 --format='%H %s'
          # 別の実行が先に push していたら取り込み直す(同じファイルは同じ内容なのでぶつからない)
          for i in 1 2 3; do
            git push -q origin HEAD:main && exit 0
            sleep $((i * 5))
            git pull -q --rebase origin main
          done
          echo "::error::mj-logs への push に3回失敗しました"
          exit 1
````

# マージ済み(origin/cloudflare の祖先)で、先頭のコミットが24時間より前の work/* ブランチを削除する(#440)。
# クラウドのセッションからはブランチを削除できないため。判定は scripts/delete_merged_branches.py。
# 削除したブランチ名と先頭の SHA は実行ログと実行サマリに残る。

on:
  workflow_dispatch:
    inputs:
      dry_run:
        description: '削除せず、対象の一覧だけ出す'
        required: true
        default: true
        type: boolean

  schedule:

    - cron: '53 22 * * *'

permissions:
  contents: write

concurrency:
  group: delete-merged-branches
  cancel-in-progress: false

jobs:
  delete:
    runs-on: ubuntu-latest
    env:
      # 毎日の実行で実際に削除するか。'false' なら schedule も dry-run で走る
      # (docs/notes/branch-operations.md「ワークフローを変更したとき」。cloudflare で手動実行して確かめてから 'true' にする)
      SCHEDULE_ENABLED: 'false'
      INPUT_DRY_RUN: ${{ inputs.dry_run }}
    steps:
      - name: dry-run かどうかを決める
        run: |
          # 手動実行は入力に従う。schedule は inputs が空になるため SCHEDULE_ENABLED で決める
          if [ "${{ github.event_name }}" = "workflow_dispatch" ]; then
            DRY_RUN="$INPUT_DRY_RUN"
          elif [ "$SCHEDULE_ENABLED" = "true" ]; then
            DRY_RUN=false
          else
            DRY_RUN=true
          fi
          echo "DRY_RUN=$DRY_RUN" >> "$GITHUB_ENV"
          echo "dry_run: $DRY_RUN"

      - name: リポジトリをチェックアウト
        uses: actions/checkout@v4
        with:
          # マージ済みの判定に完全な履歴と全ブランチが要る
          fetch-depth: 0

      - name: マージ済みの work/* を削除
        run: |
          set -o pipefail
          ARGS=""
          if [ "$DRY_RUN" = "true" ]; then
            ARGS="--dry-run"
          fi
          python3 scripts/delete_merged_branches.py $ARGS | tee summary.txt

      - name: 実行サマリに表示
        if: always()
        run: |
          {
            echo "## マージ済みの作業ブランチの削除（dry_run: $DRY_RUN）"
            echo '```'
            cat summary.txt 2>/dev/null || echo "(出力なし)"
            echo '```'
          } >> "$GITHUB_STEP_SUMMARY"
````

`scripts/delete_merged_branches.py`

````python
"""マージ済みの work/* ブランチを削除する（#440、delete-merged-branches.yml から呼ぶ）。

対象は、先頭が origin/cloudflare の祖先で、先頭のコミットが GRACE_HOURS より前のものだけ
（作業中のブランチを消さないための猶予）。削除したブランチ名と先頭の SHA を出力する
（docs/notes/branch-operations.md「ブランチを削除するとき」の記録に当たる）。
完全な履歴（fetch-depth: 0）のクローンで実行する。
"""
import argparse
import subprocess
import sys
import time

BASE = 'origin/cloudflare'
REMOTE_PREFIX = 'refs/remotes/origin/work/'
GRACE_HOURS = 24


def git(*args):
    return subprocess.run(['git', *args], check=True, capture_output=True, text=True).stdout


def is_ancestor(sha, base):
    return subprocess.run(['git', 'merge-base', '--is-ancestor', sha, base]).returncode == 0


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--dry-run', action='store_true', help='削除せず対象を出力するだけ')
    args = parser.parse_args()

    if git('rev-parse', '--is-shallow-repository').strip() == 'true':
        sys.exit('浅いクローンではマージ済みを判定できない（fetch-depth: 0 で取得する）')

    now = time.time()
    refs = git('for-each-ref', '--format=%(refname) %(objectname) %(committerdate:unix)', REMOTE_PREFIX)
    failed = False
    for line in refs.splitlines():
        ref, sha, committed = line.split()
        branch = ref.removeprefix('refs/remotes/origin/')
        age_hours = (now - int(committed)) / 3600
        if not is_ancestor(sha, BASE):
            print(f'未マージ（残す）: {branch} {sha}')
            continue
        if age_hours < GRACE_HOURS:
            print(f'猶予中（残す）: {branch} {sha} 先頭から{age_hours:.1f}時間')
            continue
        if args.dry_run:
            print(f'削除対象（dry-run）: {branch} {sha} マージ済み 先頭から{age_hours:.1f}時間')
            continue
        # 判定後に push されていたら削除しない
        result = subprocess.run(
            ['git', 'push', f'--force-with-lease=refs/heads/{branch}:{sha}', 'origin', f':refs/heads/{branch}'],
            capture_output=True, text=True)
        if result.returncode != 0:
            print(f'削除失敗: {branch} {sha}\n{result.stderr.strip()}')
            failed = True
            continue
        print(f'削除: {branch} {sha} マージ済み')
    if failed:
        sys.exit(1)


if __name__ == '__main__':
    main()
````
