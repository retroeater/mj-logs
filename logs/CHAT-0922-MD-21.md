`scripts/sync_guides.py`（全文）:

```python
"""ガイド文書を mj-logs の guide/<短い SHA>/ へ写し、写したログの末尾にリンクを付ける（#440、sync-logs.yml から呼ぶ）。

チャット側の取得の道具は一度読んだ URL をキャッシュから返すため、ガイド文書は変わるたびに新しいフォルダへ写す。
写すのは ALLOWED_PATTERNS に一致するパスだけ（パスは保つ）。docs/logs の各ログ・scripts・データは写さない。
guide/HISTORY に写したフォルダ名を古い順に1行ずつ記録し、最後の行が最新。KEEP 個より古いフォルダは消す。
mj のクローンの中で、mj-logs をチェックアウトしたディレクトリを --dest に渡して実行する。
"""
import argparse
import pathlib
import re
import shutil
import subprocess

ALLOWED_PATTERNS = [
    re.compile(r'^CLAUDE\.md$'),
    re.compile(r'^docs/handover\.md$'),
    re.compile(r'^docs/instruction-template\.md$'),
    re.compile(r'^docs/logs/_template\.md$'),
    re.compile(r'^docs/notes/[^/]+\.md$'),
]
GUIDE_DIR = 'guide'
HISTORY = 'HISTORY'
KEEP = 10
SHORT_SHA_LEN = 8
PUBLIC_BASE = 'https://github.com/retroeater/mj-logs'
# ログの末尾から直接リンクする文書
LINKED_DOCS = ['CLAUDE.md', 'docs/handover.md', 'docs/notes/chat-side-operations.md']
FOOTER_MARK = '<!-- guide-links -->'


def git(*args):
    return subprocess.run(['git', *args], check=True, capture_output=True, text=True).stdout


def is_guide(path):
    return any(p.match(path) for p in ALLOWED_PATTERNS)


def guide_files(rev):
    return [p for p in git('ls-tree', '-r', '--name-only', rev).splitlines() if is_guide(p)]


def changed_guides(base, after):
    names = git('diff', '--no-renames', '--name-only', base, after).splitlines()
    return [p for p in names if is_guide(p)]


def read_history(dest):
    path = dest / GUIDE_DIR / HISTORY
    if not path.exists():
        return []
    return [line for line in path.read_text().splitlines() if line]


def copy_guides(dest, after):
    """after の時点のガイド文書を guide/<短い SHA>/ へ写し、古いフォルダを消す。写したフォルダ名を返す。"""
    name = git('rev-parse', after).strip()[:SHORT_SHA_LEN]
    root = dest / GUIDE_DIR / name
    if root.exists():
        shutil.rmtree(root)
    files = guide_files(after)
    for path in files:
        out = root / path
        out.parent.mkdir(parents=True, exist_ok=True)
        out.write_bytes(subprocess.run(['git', 'show', f'{after}:{path}'], check=True, capture_output=True).stdout)
        print(f'guide: {path}')
    history = [h for h in read_history(dest) if h != name] + [name]
    for old in history[:-KEEP]:
        shutil.rmtree(dest / GUIDE_DIR / old, ignore_errors=True)
        print(f'guide 削除: {old}')
    history = history[-KEEP:]
    (dest / GUIDE_DIR / HISTORY).write_text('\n'.join(history) + '\n')
    print(f'guide: {GUIDE_DIR}/{name}（{len(files)}件）')
    return name


def footer(name):
    lines = [
        '',
        FOOTER_MARK,
        '---',
        '',
        f'ガイド文書（この版を写した時点の最新、mj {name}）: {PUBLIC_BASE}/tree/main/{GUIDE_DIR}/{name}',
        '',
    ]
    for doc in LINKED_DOCS:
        lines.append(f'- {doc}: {PUBLIC_BASE}/blob/main/{GUIDE_DIR}/{name}/{doc}')
    return '\n'.join(lines) + '\n'


def link_logs(dest, logs):
    """写したログの末尾に、最新のガイド文書へのリンクを付ける。既に付いていれば付け替える。"""
    history = read_history(dest)
    if not history:
        print('guide が無いためリンクを付けない')
        return
    name = history[-1]
    for log in logs:
        path = pathlib.Path(log)
        text = path.read_text()
        if FOOTER_MARK in text:
            text = text[:text.index(FOOTER_MARK)].rstrip('\n') + '\n'
        path.write_text(text + footer(name))
        print(f'link: {path.name} → {name}')


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--dest', required=True, help='mj-logs をチェックアウトしたディレクトリ')
    sub = parser.add_subparsers(dest='cmd', required=True)
    copy = sub.add_parser('copy', help='ガイド文書が変わっていれば写す')
    copy.add_argument('--base', required=True)
    copy.add_argument('--after', required=True)
    copy.add_argument('--list', action='store_true', help='写さず、写すファイルの一覧だけ出す')
    link = sub.add_parser('link', help='写したログにリンクを付ける')
    link.add_argument('logs', nargs='*')
    args = parser.parse_args()
    dest = pathlib.Path(args.dest)

    if args.cmd == 'link':
        link_logs(dest, args.logs)
        return
    if args.list:
        print('\n'.join(guide_files(args.after)))
        return
    # 初回（guide/HISTORY が無い）は変更が無くても写す
    if read_history(dest) and not changed_guides(args.base, args.after):
        print('ガイド文書の変更なし')
        return
    copy_guides(dest, args.after)


if __name__ == '__main__':
    main()
```
